uniform 	vec4 _ProjectionParams;
uniform 	vec4 hlslcc_mtx4x4unity_MatrixVP[4];
in highp vec3 in_POSITION0;
in highp vec2 in_TEXCOORD0;
out highp vec2 vs_TEXCOORD0;
out mediump vec2 vs_TEXCOORD1;
vec4 u_xlat0;
void main()
{
    u_xlat0 = in_POSITION0.yyyy * hlslcc_mtx4x4unity_MatrixVP[1];
    u_xlat0 = hlslcc_mtx4x4unity_MatrixVP[0] * in_POSITION0.xxxx + u_xlat0;
    u_xlat0 = hlslcc_mtx4x4unity_MatrixVP[2] * in_POSITION0.zzzz + u_xlat0;
    u_xlat0 = u_xlat0 + hlslcc_mtx4x4unity_MatrixVP[3];
    gl_Position = u_xlat0;
    u_xlat0.xyz = u_xlat0.xyw / u_xlat0.www;
    vs_TEXCOORD0.xy = in_TEXCOORD0.xy;
    u_xlat0.y = u_xlat0.y * _ProjectionParams.x;
    u_xlat0.xzw = u_xlat0.xzy * vec3(0.5, 0.5, 0.5);
    u_xlat0.xy = u_xlat0.zz + u_xlat0.xw;
    vs_TEXCOORD1.xy = u_xlat0.xy;
    return;
}

#ifdef FRAGMENT
#version 300 es

precision highp float;
precision highp int;
#define HLSLCC_ENABLE_UNIFORM_BUFFERS 1
#if HLSLCC_ENABLE_UNIFORM_BUFFERS
#define UNITY_UNIFORM
#else
#define UNITY_UNIFORM uniform
#endif
#define UNITY_SUPPORTS_UNIFORM_LOCATION 1
#if UNITY_SUPPORTS_UNIFORM_LOCATION
#define UNITY_LOCATION(x) layout(location = x)
#define UNITY_BINDING(x) layout(binding = x, std140)
#else
#define UNITY_LOCATION(x)
#define UNITY_BINDING(x) layout(std140)
#endif
uniform 	vec2 _GlobalMipBias;
uniform 	vec4 _Time;
uniform 	mediump float _MainTile;
uniform 	mediump float _MoveSpeed;
uniform 	mediump float _NoiseTile;
uniform 	float _NoiseSpeed;
uniform 	float _DisturbStrength;
uniform 	mediump float _Start;
uniform 	mediump float _LightAtten;
uniform 	mediump vec2 _ShapeLightBlendFactors0;
uniform 	mediump vec4 _ShapeLightMaskFilter0;
uniform 	mediump vec4 _ShapeLightInvertedFilter0;
uniform 	mediump float _HDREmulationScale;
uniform 	mediump float _UseSceneLighting;
UNITY_LOCATION(0) uniform mediump sampler2D _MainTex;
UNITY_LOCATION(1) uniform mediump sampler2D _NoiseTex;
UNITY_LOCATION(2) uniform mediump sampler2D _ShapeLightTexture0;
in highp vec2 vs_TEXCOORD0;
in mediump vec2 vs_TEXCOORD1;
layout(location = 0) out mediump vec4 SV_Target0;
vec3 u_xlat0;
mediump vec4 u_xlat16_0;
vec2 u_xlat1;
mediump vec4 u_xlat16_1;
bool u_xlatb1;
mediump vec4 u_xlat16_2;
mediump vec3 u_xlat16_3;
float u_xlat8;
bool u_xlatb13;
void main()
{
    u_xlat0.xy = vs_TEXCOORD0.xx * vec2(_NoiseTile, _MainTile);
    u_xlat1.x = 0.0;
    u_xlat1.y = _Time.y * _NoiseSpeed;
    u_xlat0.z = 0.5;
    u_xlat0.xz = u_xlat0.xz + u_xlat1.xy;
    u_xlat16_0.x = texture(_NoiseTex, u_xlat0.xz, _GlobalMipBias.x).x;
    u_xlat8 = (-vs_TEXCOORD0.y) + _Start;
    u_xlat8 = clamp(u_xlat8, 0.0, 1.0);
    u_xlat0.x = u_xlat16_0.x * u_xlat8;
    u_xlat1.y = u_xlat0.x * _DisturbStrength + vs_TEXCOORD0.y;
    u_xlat1.x = _MoveSpeed * _Time.y + u_xlat0.y;
    u_xlat16_0 = texture(_MainTex, u_xlat1.xy, _GlobalMipBias.x);
    u_xlatb1 = u_xlat16_0.w==0.0;
    if(u_xlatb1){discard;}
    u_xlat16_1.xyz = texture(_ShapeLightTexture0, vs_TEXCOORD1.xy, _GlobalMipBias.x).xyz;
    u_xlat16_2.x = dot(_ShapeLightMaskFilter0, _ShapeLightMaskFilter0);
    u_xlatb13 = u_xlat16_2.x!=0.0;
    u_xlat16_2 = (-_ShapeLightInvertedFilter0) + vec4(1.0, 1.0, 1.0, 1.0);
    u_xlat16_2.x = dot(u_xlat16_2, _ShapeLightMaskFilter0);
    u_xlat16_2.xyz = u_xlat16_1.xyz * u_xlat16_2.xxx;
    u_xlat16_2.xyz = (bool(u_xlatb13)) ? u_xlat16_2.xyz : u_xlat16_1.xyz;
    u_xlat16_3.xyz = u_xlat16_2.xyz * vec3(_ShapeLightBlendFactors0.x, _ShapeLightBlendFactors0.x, _ShapeLightBlendFactors0.x);
    u_xlat16_2.xyz = u_xlat16_2.xyz * vec3(_ShapeLightBlendFactors0.y, _ShapeLightBlendFactors0.y, _ShapeLightBlendFactors0.y);
    u_xlat16_2.xyz = u_xlat16_0.xyz * u_xlat16_3.xyz + u_xlat16_2.xyz;
    u_xlat16_1.xyz = vec3(_HDREmulationScale) * u_xlat16_2.xyz + (-u_xlat16_0.xyz);
    u_xlat16_1.w = 0.0;
    u_xlat16_1 = vec4(vec4(_UseSceneLighting, _UseSceneLighting, _UseSceneLighting, _UseSceneLighting)) * u_xlat16_1 + u_xlat16_0;
    u_xlat16_1 = max(u_xlat16_1, vec4(0.0, 0.0, 0.0, 0.0));
    u_xlat16_0 = u_xlat16_0 + (-u_xlat16_1);
    SV_Target0 = vec4(vec4(_LightAtten, _LightAtten, _LightAtten, _LightAtten)) * u_xlat16_0 + u_xlat16_1;
    return;
}
