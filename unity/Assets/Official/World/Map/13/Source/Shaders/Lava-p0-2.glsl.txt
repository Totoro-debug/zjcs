precision highp float;

uniform 	vec2 _GlobalMipBias;
uniform 	vec4 _Time;
uniform 	mediump float _MainTile;
uniform 	mediump float _MoveSpeed;
uniform 	mediump float _NoiseTile;
uniform 	float _NoiseSpeed;
uniform 	float _Fluctuation;
uniform 	float _DisturbStrength;
uniform 	mediump vec2 _ShapeLightBlendFactors0;
uniform 	mediump vec4 _ShapeLightMaskFilter0;
uniform 	mediump vec4 _ShapeLightInvertedFilter0;
uniform 	mediump float _HDREmulationScale;
uniform 	mediump float _UseSceneLighting;
UNITY_LOCATION(0) uniform mediump sampler2D _MainTex;
UNITY_LOCATION(1) uniform mediump sampler2D _NoiseTex;
UNITY_LOCATION(2) uniform mediump sampler2D _ShapeLightTexture0;
in highp vec2 vs_TEXCOORD2;
in highp vec3 vs_TEXCOORD1;
in mediump vec2 vs_TEXCOORD3;
layout(location = 0) out mediump vec4 SV_Target0;
vec4 u_xlat0;
mediump vec4 u_xlat16_0;
vec2 u_xlat1;
mediump vec4 u_xlat16_1;
bool u_xlatb1;
mediump vec4 u_xlat16_2;
mediump vec3 u_xlat16_3;
vec2 u_xlat4;
bool u_xlatb13;
void main()
{
    u_xlat0 = vs_TEXCOORD1.xyxy * vec4(_NoiseTile, _NoiseTile, _MainTile, _MainTile);
    u_xlat0.xy = _Time.yy * vec2(_NoiseSpeed) + u_xlat0.xy;
    u_xlat16_0.x = texture(_NoiseTex, u_xlat0.xy, _GlobalMipBias.x).x;
    u_xlat1.xy = vs_TEXCOORD2.xy * vec2(vec2(_MoveSpeed, _MoveSpeed));
    u_xlat4.xy = u_xlat1.xy * _Time.yy + u_xlat0.zw;
    u_xlat0.xy = u_xlat16_0.xx * vec2(vec2(_DisturbStrength, _DisturbStrength)) + u_xlat4.xy;
    u_xlat16_0 = texture(_MainTex, u_xlat0.xy, _GlobalMipBias.x);
    u_xlat1.x = sin(_Time.y);
    u_xlat1.x = u_xlat1.x * _Fluctuation + 1.0;
    u_xlat0 = u_xlat16_0 * u_xlat1.xxxx;
    u_xlatb1 = u_xlat0.w==0.0;
    if(u_xlatb1){discard;}
    u_xlat16_1.xyz = texture(_ShapeLightTexture0, vs_TEXCOORD3.xy, _GlobalMipBias.x).xyz;
    u_xlat16_2.x = dot(_ShapeLightMaskFilter0, _ShapeLightMaskFilter0);
    u_xlatb13 = u_xlat16_2.x!=0.0;
    u_xlat16_2 = (-_ShapeLightInvertedFilter0) + vec4(1.0, 1.0, 1.0, 1.0);
    u_xlat16_2.x = dot(u_xlat16_2, _ShapeLightMaskFilter0);
    u_xlat16_2.xyz = u_xlat16_1.xyz * u_xlat16_2.xxx;
    u_xlat16_2.xyz = (bool(u_xlatb13)) ? u_xlat16_2.xyz : u_xlat16_1.xyz;
    u_xlat16_3.xyz = u_xlat16_2.xyz * _ShapeLightBlendFactors0.xxx;
    u_xlat16_2.xyz = u_xlat16_2.xyz * _ShapeLightBlendFactors0.yyy;
    u_xlat16_2.xyz = u_xlat0.xyz * u_xlat16_3.xyz + u_xlat16_2.xyz;
    u_xlat16_1.xyz = vec3(_HDREmulationScale) * u_xlat16_2.xyz + (-u_xlat0.xyz);
    u_xlat16_1.w = 0.0;
    u_xlat16_0 = vec4(vec4(_UseSceneLighting, _UseSceneLighting, _UseSceneLighting, _UseSceneLighting)) * u_xlat16_1 + u_xlat0;
    SV_Target0 = max(u_xlat16_0, vec4(0.0, 0.0, 0.0, 0.0));
    return;
}
