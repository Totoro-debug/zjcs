precision highp float;
precision highp int;
#define HLSLCC_ENABLE_UNIFORM_BUFFERS 1

#else
#define UNITY_UNIFORM uniform

#define UNITY_SUPPORTS_UNIFORM_LOCATION 1
#if UNITY_SUPPORTS_UNIFORM_LOCATION
#define UNITY_LOCATION(x) layout(location = x)
#define UNITY_BINDING(x) layout(binding = x, std140)
#else
#define UNITY_LOCATION(x)
#define UNITY_BINDING(x) layout(std140)
#endif
uniform 	vec2 _GlobalMipBias;
uniform 	float _ShadowVolumeIntensity;
uniform 	mediump vec4 _ShadowColorMask;
UNITY_LOCATION(0) uniform mediump sampler2D _CookieTex;
UNITY_LOCATION(1) uniform mediump sampler2D _ShadowTex;
in mediump vec4 vs_COLOR0;
in mediump vec2 vs_TEXCOORD0;
in highp vec2 vs_TEXCOORD1;
layout(location = 0) out mediump vec4 SV_Target0;
vec4 u_xlat0;
mediump vec4 u_xlat16_0;
vec4 u_xlat1;
mediump vec4 u_xlat16_1;
bool u_xlatb1;
mediump float u_xlat16_2;
mediump float u_xlat16_5;
void main()
{
    u_xlat16_0 = texture(_CookieTex, vs_TEXCOORD0.xy, _GlobalMipBias.x);
    u_xlat0 = u_xlat16_0 * vs_COLOR0;
    u_xlatb1 = _ShadowVolumeIntensity<1.0;
    if(u_xlatb1){
        u_xlat16_1 = texture(_ShadowTex, vs_TEXCOORD1.xy, _GlobalMipBias.x);
        u_xlat1 = u_xlat16_1;
        u_xlat1 = clamp(u_xlat1, 0.0, 1.0);
        u_xlat16_2 = dot(_ShadowColorMask, u_xlat1);
        u_xlat16_2 = (-u_xlat16_2) + 1.0;
        u_xlat1.xyz = u_xlat0.xyz * vec3(vec3(_ShadowVolumeIntensity, _ShadowVolumeIntensity, _ShadowVolumeIntensity));
        u_xlat16_5 = (-u_xlat16_2) + 1.0;
        u_xlat1.xyz = u_xlat1.xyz * vec3(u_xlat16_5);
        u_xlat1.xyz = u_xlat0.xyz * vec3(u_xlat16_2) + u_xlat1.xyz;
        u_xlat16_1.xyz = u_xlat1.xyz;
    } else {
        u_xlat16_1.xyz = u_xlat0.xyz;
    }
    SV_Target0.xyz = u_xlat16_1.xyz;
    SV_Target0.w = u_xlat0.w;
    return;
}
