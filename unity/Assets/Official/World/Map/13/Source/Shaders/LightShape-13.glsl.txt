in highp vec3 in_POSITION0;
in highp vec4 in_COLOR0;
in highp vec2 in_TEXCOORD0;

out mediump vec2 vs_TEXCOORD0;
out highp vec2 vs_TEXCOORD1;
out mediump vec4 vs_TEXCOORD2;
out mediump vec2 vs_TEXCOORD3;
vec4 u_xlat0;
vec4 u_xlat1;
vec3 u_xlat2;
float u_xlat9;
void main()
{
    u_xlat0.xy = vec2(_FalloffDistance) * in_COLOR0.xy + in_POSITION0.xy;
    u_xlat1 = u_xlat0.yyyy * hlslcc_mtx4x4unity_ObjectToWorld[1];
    u_xlat0 = hlslcc_mtx4x4unity_ObjectToWorld[0] * u_xlat0.xxxx + u_xlat1;
    u_xlat0 = hlslcc_mtx4x4unity_ObjectToWorld[2] * in_POSITION0.zzzz + u_xlat0;
    u_xlat0 = u_xlat0 + hlslcc_mtx4x4unity_ObjectToWorld[3];
    u_xlat1 = u_xlat0.yyyy * hlslcc_mtx4x4unity_MatrixVP[1];
    u_xlat1 = hlslcc_mtx4x4unity_MatrixVP[0] * u_xlat0.xxxx + u_xlat1;
    u_xlat1 = hlslcc_mtx4x4unity_MatrixVP[2] * u_xlat0.zzzz + u_xlat1;
    u_xlat1 = hlslcc_mtx4x4unity_MatrixVP[3] * u_xlat0.wwww + u_xlat1;
    gl_Position = u_xlat1;
    u_xlat1.xy = u_xlat1.xy / u_xlat1.ww;
    u_xlat1.xy = u_xlat1.xy * vec2(0.5, 0.5) + vec2(0.5, 0.5);
    vs_COLOR0.xyz = vec3(_InverseHDREmulationScale) * _LightColor.xyz;
    vs_COLOR0.w = in_COLOR0.w;
    vs_TEXCOORD0.xy = in_TEXCOORD0.xy;
    vs_TEXCOORD1.xy = u_xlat1.xy;
    vs_TEXCOORD3.xy = u_xlat1.xy;
    u_xlat1.xyz = (-u_xlat0.xyz) + _LightPosition.xyz;
    u_xlat2.x = hlslcc_mtx4x4unity_MatrixV[0].z;
    u_xlat2.y = hlslcc_mtx4x4unity_MatrixV[1].z;
    u_xlat2.z = hlslcc_mtx4x4unity_MatrixV[2].z;
    u_xlat9 = dot(u_xlat1.xyz, u_xlat2.xyz);
    u_xlat9 = u_xlat9 + (-_LightZDistance);
    u_xlat1.xyz = (-vec3(u_xlat9)) * u_xlat2.xyz + _LightPosition.xyz;
    u_xlat0.xyz = (-u_xlat0.xyz) + u_xlat1.xyz;
    vs_TEXCOORD2.xyz = u_xlat0.xyz;
    vs_TEXCOORD2.w = 0.0;
    return;
}

uniform 	vec2 _GlobalMipBias;
uniform 	float _ShadowIntensity;
uniform 	mediump vec4 _ShadowColorMask;
UNITY_LOCATION(0) uniform mediump sampler2D _CookieTex;
UNITY_LOCATION(1) uniform mediump sampler2D _NormalMap;
UNITY_LOCATION(2) uniform mediump sampler2D _ShadowTex;
in mediump vec4 vs_COLOR0;
in mediump vec2 vs_TEXCOORD0;
in highp vec2 vs_TEXCOORD1;
in mediump vec4 vs_TEXCOORD2;
in mediump vec2 vs_TEXCOORD3;
layout(location = 0) out mediump vec4 SV_Target0;
vec3 u_xlat0;
mediump vec4 u_xlat16_0;
vec4 u_xlat1;
mediump vec4 u_xlat16_1;
bool u_xlatb1;
mediump vec3 u_xlat16_2;
mediump vec3 u_xlat16_3;
mediump float u_xlat16_6;
mediump float u_xlat16_14;
void main()
{
    u_xlat16_0 = texture(_CookieTex, vs_TEXCOORD0.xy, _GlobalMipBias.x);
    u_xlat16_0 = u_xlat16_0.wwww * u_xlat16_0;
    u_xlat16_0 = u_xlat16_0 * vs_COLOR0;
    u_xlat16_1.xyz = texture(_NormalMap, vs_TEXCOORD3.xy, _GlobalMipBias.x).xyz;
    u_xlat16_2.xyz = u_xlat16_1.xyz * vec3(2.0, 2.0, 2.0) + vec3(-1.0, -1.0, -1.0);
    u_xlat16_14 = dot(vs_TEXCOORD2.xyz, vs_TEXCOORD2.xyz);
    u_xlat16_14 = inversesqrt(u_xlat16_14);
    u_xlat16_3.xyz = vec3(u_xlat16_14) * vs_TEXCOORD2.xyz;
    u_xlat16_2.x = dot(u_xlat16_3.xyz, u_xlat16_2.xyz);
    u_xlat16_2.x = clamp(u_xlat16_2.x, 0.0, 1.0);
    u_xlat16_0 = u_xlat16_0 * u_xlat16_2.xxxx;
    u_xlatb1 = _ShadowIntensity<1.0;
    if(u_xlatb1){
        u_xlat16_1 = texture(_ShadowTex, vs_TEXCOORD1.xy, _GlobalMipBias.x);
        u_xlat1 = u_xlat16_1;
        u_xlat1 = clamp(u_xlat1, 0.0, 1.0);
        u_xlat16_2.x = dot(_ShadowColorMask, u_xlat1);
        u_xlat16_2.x = (-u_xlat16_2.x) + 1.0;
        u_xlat1.xyz = u_xlat16_0.xyz * vec3(_ShadowIntensity);
        u_xlat16_6 = (-u_xlat16_2.x) + 1.0;
        u_xlat1.xyz = u_xlat1.xyz * vec3(u_xlat16_6);
        u_xlat0.xyz = u_xlat16_0.xyz * u_xlat16_2.xxx + u_xlat1.xyz;
        u_xlat16_0.xyz = u_xlat0.xyz;
    }
    SV_Target0 = u_xlat16_0;
    return;
}
