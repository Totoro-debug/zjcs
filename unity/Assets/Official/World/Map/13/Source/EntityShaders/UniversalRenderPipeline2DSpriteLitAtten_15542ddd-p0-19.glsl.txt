precision highp float;
precision highp int;
#define HLSLCC_ENABLE_UNIFORM_BUFFERS 1
#if HLSLCC_ENABLE_UNIFORM_BUFFERS
#define UNITY_UNIFORM
#else

#define UNITY_SUPPORTS_UNIFORM_LOCATION 1

#endif
uniform 	vec2 _GlobalMipBias;
uniform 	float _LightAtten;
uniform 	mediump vec2 _ShapeLightBlendFactors0;
uniform 	mediump vec4 _ShapeLightMaskFilter0;
uniform 	mediump vec4 _ShapeLightInvertedFilter0;
uniform 	mediump vec2 _ShapeLightBlendFactors1;
uniform 	mediump vec4 _ShapeLightMaskFilter1;
uniform 	mediump vec4 _ShapeLightInvertedFilter1;
uniform 	mediump float _HDREmulationScale;
uniform 	mediump float _UseSceneLighting;
UNITY_LOCATION(0) uniform mediump sampler2D _MainTex;
UNITY_LOCATION(1) uniform mediump sampler2D _MaskTex;
UNITY_LOCATION(2) uniform mediump sampler2D _ShapeLightTexture0;
UNITY_LOCATION(3) uniform mediump sampler2D _ShapeLightTexture1;
in mediump vec4 vs_COLOR0;
in highp vec2 vs_TEXCOORD0;
in mediump vec2 vs_TEXCOORD1;
layout(location = 0) out mediump vec4 SV_Target0;
vec3 u_xlat0;
mediump vec4 u_xlat16_0;
vec4 u_xlat1;
mediump vec4 u_xlat16_2;
mediump vec3 u_xlat16_3;
mediump vec4 u_xlat16_4;
mediump vec4 u_xlat16_5;
mediump vec4 u_xlat16_6;
bool u_xlatb21;
mediump float u_xlat16_25;
void main()
{
    u_xlat16_0 = texture(_MainTex, vs_TEXCOORD0.xy, _GlobalMipBias.x);
    u_xlat1 = u_xlat16_0 * vs_COLOR0;
    u_xlatb21 = u_xlat1.w==0.0;
    if(u_xlatb21){discard;}
    u_xlat16_2 = texture(_MaskTex, vs_TEXCOORD0.xy, _GlobalMipBias.x);
    u_xlat16_3.xyz = texture(_ShapeLightTexture0, vs_TEXCOORD1.xy, _GlobalMipBias.x).xyz;
    u_xlat16_4.x = dot(_ShapeLightMaskFilter0, _ShapeLightMaskFilter0);
    u_xlatb21 = u_xlat16_4.x!=0.0;
    u_xlat16_4 = (-_ShapeLightInvertedFilter0) + vec4(1.0, 1.0, 1.0, 1.0);
    u_xlat16_5 = (-u_xlat16_2) + vec4(1.0, 1.0, 1.0, 1.0);
    u_xlat16_6 = u_xlat16_5 * _ShapeLightInvertedFilter0;
    u_xlat16_4 = u_xlat16_4 * u_xlat16_2 + u_xlat16_6;
    u_xlat16_4.x = dot(u_xlat16_4, _ShapeLightMaskFilter0);
    u_xlat16_4.xyz = u_xlat16_3.xyz * u_xlat16_4.xxx;
    u_xlat16_4.xyz = (bool(u_xlatb21)) ? u_xlat16_4.xyz : u_xlat16_3.xyz;
    u_xlat16_3.xyz = texture(_ShapeLightTexture1, vs_TEXCOORD1.xy, _GlobalMipBias.x).xyz;
    u_xlat16_25 = dot(_ShapeLightMaskFilter1, _ShapeLightMaskFilter1);
    u_xlatb21 = u_xlat16_25!=0.0;
    u_xlat16_6 = (-_ShapeLightInvertedFilter1) + vec4(1.0, 1.0, 1.0, 1.0);
    u_xlat16_5 = u_xlat16_5 * _ShapeLightInvertedFilter1;
    u_xlat16_2 = u_xlat16_6 * u_xlat16_2 + u_xlat16_5;
    u_xlat16_25 = dot(u_xlat16_2, _ShapeLightMaskFilter1);
    u_xlat16_5.xyz = u_xlat16_3.xyz * vec3(u_xlat16_25);
    u_xlat16_5.xyz = (bool(u_xlatb21)) ? u_xlat16_5.xyz : u_xlat16_3.xyz;
    u_xlat16_6.xyz = u_xlat16_5.xyz * _ShapeLightBlendFactors1.xxx;
    u_xlat16_5.xyz = u_xlat16_5.xyz * _ShapeLightBlendFactors1.yyy;
    u_xlat16_6.xyz = u_xlat16_4.xyz * _ShapeLightBlendFactors0.xxx + u_xlat16_6.xyz;
    u_xlat16_4.xyz = u_xlat16_4.xyz * _ShapeLightBlendFactors0.yyy + u_xlat16_5.xyz;
    u_xlat16_4.xyz = u_xlat1.xyz * u_xlat16_6.xyz + u_xlat16_4.xyz;
    u_xlat16_4.xyz = vec3(_HDREmulationScale) * u_xlat16_4.xyz + (-u_xlat1.xyz);
    u_xlat16_4.xyz = vec3(vec3(_UseSceneLighting, _UseSceneLighting, _UseSceneLighting)) * u_xlat16_4.xyz + u_xlat1.xyz;
    u_xlat16_4.xyz = max(u_xlat16_4.xyz, vec3(0.0, 0.0, 0.0));
    u_xlat0.xyz = vs_COLOR0.xyz * u_xlat16_0.xyz + (-u_xlat16_4.xyz);
    u_xlat1.xyz = vec3(_LightAtten) * u_xlat0.xyz + u_xlat16_4.xyz;
    SV_Target0 = u_xlat1;
    return;
}
